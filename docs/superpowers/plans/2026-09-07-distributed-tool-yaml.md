# Distributed Tool YAML Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move per-tool metadata from central `data/*/static.yaml` files into individual `.tool.yaml` files in each tool's GitHub repo, with a `data/categories.yaml` registry controlling groupings, and a new fetch script assembling the final `static.yaml` at CI time.

**Architecture:** A new `scripts/fetch_tool_metadata.py` reads `data/categories.yaml`, fetches `.tool.yaml` from each tool repo's raw GitHub URL, merges in website-side fields (`id`, `group`, `no_sap`), and writes `data/<section>/static.yaml` in the same format consumed by Hugo and `fetch_releases.py` today. The Hugo template gets one change: `.femto` → `.self` in the `vs` comparison table (plus optional `self_label` column header override). Everything else is unchanged.

**Tech Stack:** Python 3.x, PyYAML, requests, Hugo, GitHub Actions

```python
#!/usr/bin/env python3
"""
Reads  data/categories.yaml
Writes data/<section>/static.yaml  for each collection

Fetches .tool.yaml from each tool repo's raw GitHub URL and merges in
website-side fields (id, group, no_sap).

Exit codes:
  0 — success, even if some tools were skipped due to warnings
  1 — data/categories.yaml is missing or unparseable
"""

import argparse
import os
import sys
from pathlib import Path

import requests
import yaml

ROOT = Path(__file__).resolve().parent.parent
GITHUB_TOKEN = os.environ.get("GITHUB_TOKEN", "")

REQUIRED_FIELDS = {"tag", "tagline", "tagline_short"}

SCHEMA_HEADER = """\
# ─────────────────────────────────────────────────────────────────────────────
# AUTO-GENERATED by scripts/fetch_tool_metadata.py — do not edit by hand.
# Edit .tool.yaml in each tool's repo, or data/categories.yaml for grouping.
# ─────────────────────────────────────────────────────────────────────────────

"""


def _raw_url(repo_url: str, filename: str) -> str:
    """Convert a GitHub repo URL to a raw.githubusercontent.com URL for a file."""
    # https://github.com/owner/repo -> https://raw.githubusercontent.com/owner/repo/HEAD/filename
    parts = repo_url.rstrip("/").replace("https://github.com/", "")
    return f"https://raw.githubusercontent.com/{parts}/HEAD/{filename}"


def _fetch_tool_yaml(repo_url: str) -> dict | None:
    """
    Fetch and parse .tool.yaml from a GitHub repo.
    Returns None on any failure, printing a warning.
    """
    url = _raw_url(repo_url, ".tool.yaml")
    headers = {}
    if GITHUB_TOKEN:
        headers["Authorization"] = f"token {GITHUB_TOKEN}"
    try:
        resp = requests.get(url, headers=headers, timeout=15)
        if resp.status_code == 404:
            print(f"  WARNING: .tool.yaml not found in {repo_url}", file=sys.stderr)
            return None
        resp.raise_for_status()
    except requests.RequestException as exc:
        print(f"  WARNING: network error fetching {repo_url}: {exc}", file=sys.stderr)
        return None

    try:
        data = yaml.safe_load(resp.text)
    except yaml.YAMLError as exc:
        print(f"  WARNING: YAML parse error in {repo_url}/.tool.yaml: {exc}", file=sys.stderr)
        return None

    if not isinstance(data, dict):
        print(f"  WARNING: .tool.yaml in {repo_url} is not a YAML mapping", file=sys.stderr)
        return None

    missing = REQUIRED_FIELDS - data.keys()
    if missing:
        print(
            f"  WARNING: .tool.yaml in {repo_url} missing required fields: {', '.join(sorted(missing))}",
            file=sys.stderr,
        )
        return None

    return data


def _repo_id(repo_url: str) -> str:
    """Last path segment of the repo URL — used as the tool id."""
    return repo_url.rstrip("/").split("/")[-1]


def _build_static_yaml(tools: list[dict]) -> str:
    """Serialise tool list to a YAML string with the auto-generated header."""

    class _LiteralStr(str):
        pass

    def _literal_representer(dumper, data):
        return dumper.represent_scalar("tag:yaml.org,2002:str", data, style="|")

    class _Dumper(yaml.Dumper):
        pass

    _Dumper.add_representer(_LiteralStr, _literal_representer)

    def _wrap(obj):
        if isinstance(obj, str):
            return _LiteralStr(obj) if "\n" in obj else obj
        if isinstance(obj, list):
            return [_wrap(i) for i in obj]
        if isinstance(obj, dict):
            return {k: _wrap(v) for k, v in obj.items()}
        return obj

    dumped = yaml.dump(
        _wrap(tools),
        Dumper=_Dumper,
        allow_unicode=True,
        default_flow_style=False,
        sort_keys=False,
        width=100,
    )
    return SCHEMA_HEADER + dumped


def fetch_section(section_id: str, collection: dict) -> list[dict]:
    """Fetch all tools for one collection, return merged tool list."""
    tools = []
    for group in collection.get("groups", []):
        group_name = group["name"]

        for tool_entry in group.get("tools", []):
            repo_url = tool_entry["repo"]
            tool_id = _repo_id(repo_url)
            data = _fetch_tool_yaml(repo_url)
            if data is None:
                continue
            merged = {
                "id": tool_id,
                "group": group_name,
                **data,
            }
            if tool_entry.get("no_sap"):
                merged["no_sap"] = True
            tools.append(merged)
            print(f"  {tool_id}: ok")

        for fallback in group.get("fallback_tools", []):
            merged = {"group": group_name, **fallback}
            tools.append(merged)
            print(f"  {fallback['id']}: ok (fallback)")

    return tools


def main():
    parser = argparse.ArgumentParser(description="Fetch tool metadata from individual repos.")
    parser.add_argument(
        "--section",
        help="Only process this section (e.g. jvm-tools). Omit to process all.",
    )
    args = parser.parse_args()

    categories_path = ROOT / "data" / "categories.yaml"
    if not categories_path.exists():
        print(f"ERROR: {categories_path} not found", file=sys.stderr)
        sys.exit(1)

    try:
        categories = yaml.safe_load(categories_path.read_text())
    except yaml.YAMLError as exc:
        print(f"ERROR: failed to parse {categories_path}: {exc}", file=sys.stderr)
        sys.exit(1)

    for collection in categories.get("collections", []):
        section_id = collection["id"]
        if args.section and section_id != args.section:
            continue

        print(f"\n[{section_id}]")
        tools = fetch_section(section_id, collection)

        out_path = ROOT / "data" / section_id / "static.yaml"
        out_path.parent.mkdir(parents=True, exist_ok=True)
        out_path.write_text(_build_static_yaml(tools))
        print(f"  Wrote {len(tools)} tools to {out_path.relative_to(ROOT)}")


if __name__ == "__main__":
    main()
```

name: Daily data update

on:
  schedule:
    - cron: '23 4 * * 1-5'  # every weekday at 04:23 UTC

jobs:
  fetch-metadata:
    name: Fetch tool metadata
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
      - uses: actions/upload-artifact@v4
        with:
          name: releases-data
          path: |
            data/femto/releases.json
            data/jvm-tools/releases.json
            data/experiments/releases.json

  blog:
    name: Fetch blog posts
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.x'

      - run: python scripts/fetch_blog.py

          name: blog-data

          name: releases-data
          path: data
          merge-multiple: true

        with:
          name: blog-data
          path: data
      - uses: actions/download-artifact@v4
        with:
          name: talks-data
          path: data
      - name: Commit if changed
        run: |
          git diff --quiet -- \
            data/femto/static.yaml \
            data/jvm-tools/static.yaml \
            data/experiments/static.yaml \
            data/femto/releases.json \
            data/jvm-tools/releases.json \
            data/experiments/releases.json \
            data/blog.json \
            data/talks.json \
          || (
            git config user.email "actions@github.com"
            git config user.name "github-actions"
            git add \
              data/femto/static.yaml \
              data/jvm-tools/static.yaml \
              data/experiments/static.yaml \
              data/femto/releases.json \
              data/jvm-tools/releases.json \
              data/experiments/releases.json \
              data/blog.json \
              data/talks.json

            git push
          )

  sync-profile-readme:

    needs: [blog, talks]
    runs-on: ubuntu-latest
    steps:

        with:
          name: blog-data
          path: data
      - uses: actions/download-artifact@v4
        with:
          name: talks-data
          path: data
      - uses: actions/setup-python@v5
        with:
          python-version: '3.x'
      - name: Generate README
        run: python scripts/gen_profile_readme.py
      - name: Push to parttimenerd/parttimenerd

          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: |
          cd /tmp

          cp $GITHUB_WORKSPACE/docs/profile-README.md profile-readme/README.md
          cd profile-readme
          git diff --quiet README.md || (
            git config user.email "actions@github.com"
            git config user.name "github-actions"
            git add README.md
            git commit -m "chore: sync profile README"
            git push
          )
